cmake_minimum_required(VERSION 3.25)
include(cmake/prelude.cmake)
project(safe_c_starter VERSION 0.1.0 LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SAFETY_PROFILE "ordinary" CACHE STRING "Protected profile selected by outer evaluator")
set(SAFETY_CASE "" CACHE STRING "Explicit qualification case; no application wildcard exclusions")
set(SAFETY_VARIANT "both" CACHE STRING "Qualification variants")
include(cmake/Safety.cmake)
enable_testing()
# Application transition requires an externally approved contract. Config flags
# alone never authorize a bootstrap bypass; the outer inventory gate checks mode.
file(GLOB_RECURSE app_sources CONFIGURE_DEPENDS "src/*.c" "include/*.c")
if(app_sources)
  message(FATAL_ERROR "bootstrap contract forbids application sources; obtain approved production inventory")
endif()
safety_add_program(infrastructure_demo tests/integration/demo.c fuzz/parser_good.c)
add_test(NAME infrastructure.boundaries COMMAND infrastructure_demo)
safety_add_program(hardening_probe tests/integration/hardening.c)
add_test(NAME infrastructure.hardening COMMAND hardening_probe)
file(READ "${PROJECT_SOURCE_DIR}/fuzz/targets.json" fuzz_manifest)
string(JSON fuzz_harness GET "${fuzz_manifest}" harness)
string(JSON fuzz_bad GET "${fuzz_manifest}" parser_bad)
string(JSON fuzz_good GET "${fuzz_manifest}" parser_good)
string(JSON fuzz_driver GET "${fuzz_manifest}" ordinary_driver)
include(safety/targets.cmake)
